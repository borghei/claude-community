var TIMES = ['03:00','04:30','06:15','13:40','15:10','17:55','21:30','23:45','02:20','09:05'];
var ACT = [
 'Begin the day with a deeply immersive visit to %s, where the vibe is, frankly, unmatched, and every angle is a story waiting to be discovered by someone exactly like you',
 'Then transition seamlessly to %s, a hidden gem that is not hidden at all, and spend as long as your energy permits absorbing the local magic',
 'Next, soak up the authentic atmosphere at %s, before a spontaneous detour to %s, which is only %s km away by taxi, a completely reasonable distance',

 'Treat yourself to an iconic stop at %s, then dive into the energy of %s, locals call it "the place"'
];
function pick(a,n){ return a[(n%a.length+a.length)%a.length]; }
function hash(str){ var h=7; for(var i=0;i<str.length;i++) h=(h*31+str.charCodeAt(i))|0; return Math.abs(h); }

function generate(msg){
  S.msg = msg; go('loading');
  setTimeout(function(){
    S.seed = hash(JSON.stringify(S.ans)) + (S.regenCount=(S.regenCount||0)+1)*13;
    var p = pick(PLACES, S.seed), days=[];
    for(var d=0; d<7; d++){
      var s = '', n = 4;
      for(var i=0;i<n;i++){
        var sa = pick(p.sites,S.seed+d*3+i), sb = pick(p.sites,S.seed+d*5+i+2);
        var tm = pick(TIMES, S.seed+d*7+i*3);
        var line = pick(ACT, S.seed+d+i).replace('%s',sa).replace('%s',sb).replace('%s',(120+((S.seed+d*91+i*37)%1900)));
        s += '<b>'+tm+'</b> &ndash; '+line+'. ';
      }
      days.push(s+'You may also wish to explore the area at your own pace, although we recommend not wasting a single minute.');
    }
    S.trip = {name:p.name,country:p.country,days:days};
    go('result');
  }, 2200);
}
